  function schedulePushNotifications() {

    if (!('Notification' in window) || Notification.permission !== 'granted') return;

    const now = new Date();

        if (!wakeupTime) return; // No wakeup set yet
        const [wh, wm] = wakeupTime.split(':').map(Number);
        const wakeDate = new Date();
        wakeDate.setHours(wh, wm, 0, 0);
        const firstFire = new Date(wakeDate.getTime() + med.offsetMins * 60000);
        if (firstFire > now) {
          const delay = firstFire - now;
          const id = setTimeout(() => fireNotification(med), delay);
          newIds.push(id);
        }
      } else {
        // Fixed time
        const [h, m] = med.time.split(':').map(Number);
        const fireDate = new Date();
        fireDate.setHours(h, m, 0, 0);
        if (fireDate <= now) fireDate.setDate(fireDate.getDate() + 1);
        const delay = fireDate - now;
        const id = setTimeout(() => fireNotification(med), delay);
        newIds.push(id);
      }
    });

    localStorage.setItem('sante_notif_ids', JSON.stringify(newIds));
    } catch(e) { console.warn('Notification scheduling error:', e); }

        iw.querySelectorAll('.freq-wheel-item').forEach((el, i) => {
          el.classList.toggle('active', i === idx);
        });
        currentIntervalHours = vals[idx];
      });
    }

    renderWakeup();
    schedulePushNotifications();
  });

  // ---- MEDS ----
  function saveMed() {
    const name = document.getElementById('medName').value.trim();
    const dose = document.getElementById('medDose').value.trim();
    if (!name) return;

    const isFreq = selectedEmoji === '🩺' || selectedEmoji === '🧦';

    if (isFreq) {
      const timeInputs = document.querySelectorAll('#dynamicTimes input[type=time]');
      const labels = selectedEmoji === '🧦'
        ? ['– Anziehen', '– Ausziehen', '– 3. Mal', '– 4. Mal']
        : ['– 1. Messung', '– 2. Messung', '– 3. Messung', '– 4. Messung'];
      timeInputs.forEach((inp, i) => {
        const suffix = timeInputs.length === 1 ? '' : ' ' + labels[i];
        meds.push({ id: Date.now() + i, emoji: selectedEmoji, name: name + suffix, dose: '', time: inp.value, reminderType: 'fixed', offsetMins: 0, intervalHours: 6 });
      });
    } else {
      const time = document.getElementById('medTime').value;
      const rType = currentReminderType;
      const offset = currentOffsetMins;
      const interval = currentIntervalHours;
      meds.push({ id: Date.now(), emoji: selectedEmoji, name, dose, time, reminderType: rType, offsetMins: offset, intervalHours: interval });
    }

    localStorage.setItem('sante_meds', JSON.stringify(meds));
    document.getElementById('modalMed').classList.remove('show');
    document.getElementById('medName').value = '';
    document.getElementById('medDose').value = '';
    schedulePushNotifications();
    render();
  }

  function deleteMed(id) {
    meds = meds.filter(m => m.id !== id);
    localStorage.setItem('sante_meds', JSON.stringify(meds));
    render();
  }

  function toggleTaken(id) {
    const idx = takenToday.indexOf(id);
    if (idx >= 0) takenToday.splice(idx, 1);
    else takenToday.push(id);
    localStorage.setItem('sante_taken_' + today(), JSON.stringify(takenToday));
    render();
  }

  function renderMeds(containerId, showDelete = false) {
    const el = document.getElementById(containerId);
    if (!meds.length) {
      el.innerHTML = `<div class="empty"><div class="empty-icon">💊</div>Noch keine Medikamente eingetragen</div>`;
      return;
    }
    const sorted = [...meds].sort((a,b) => a.time.localeCompare(b.time));
    const taken = takenToday.length;
    let html = '';
    if (containerId === 'todayMeds' && meds.length > 0) {
      html += `<div class="progress-summary">
        <div>
          <div class="progress-text">✓ ${taken} von ${meds.length} genommen</div>
          <div class="progress-sub">${taken === meds.length ? 'Alle erledigt! 🎉' : 'Weiter so!'}</div>
        </div>
      </div>`;
    }
    sorted.forEach(med => {
      const done = takenToday.includes(med.id);
      html += `<div class="med-card ${done ? 'taken' : ''}">
        <div class="med-icon">${med.emoji}</div>
        <div class="med-info">
          <div class="med-name">${med.name}</div>
          <div class="med-dose">${med.dose || ''}</div>
          <div class="med-time">🕐 ${med.time} Uhr</div>
        </div>
        <button class="check-btn ${done ? 'done' : ''}" onclick="toggleTaken(${med.id})">${done ? '✓' : ''}</button>
        ${showDelete ? `<button class="delete-btn" onclick="deleteMed(${med.id})">✕</button>` : ''}
      </div>`;
    });
    el.innerHTML = html;
  }

  // ---- APPTS ----
  function saveAppt() {
    const title = document.getElementById('apptTitle').value.trim();
    const sub = document.getElementById('apptSub').value.trim();
    const date = document.getElementById('apptDate').value;
    const time = document.getElementById('apptTime').value;
    if (!title || !date) return;
    appts.push({ id: Date.now(), title, sub, date, time });
    appts.sort((a,b) => a.date.localeCompare(b.date));
    localStorage.setItem('sante_appts', JSON.stringify(appts));
    document.getElementById('modalAppt').classList.remove('show');
    document.getElementById('apptTitle').value = '';
    document.getElementById('apptSub').value = '';
    render();
  }

  function deleteAppt(id) {
    appts = appts.filter(a => a.id !== id);
    localStorage.setItem('sante_appts', JSON.stringify(appts));
    render();
  }

  function renderAppts(containerId, limit = 999, showDelete = false) {
    const el = document.getElementById(containerId);
    const future = appts.filter(a => a.date >= today()).slice(0, limit);
    if (!future.length) {
      el.innerHTML = `<div class="empty"><div class="empty-icon">📅</div>Keine bevorstehenden Termine</div>`;
      return;
    }
    let html = '';
    future.forEach(appt => {
      const d = new Date(appt.date + 'T12:00:00');
      const diffDays = Math.ceil((d - new Date().setHours(0,0,0,0)) / 86400000);
      const soonLabel = diffDays === 0 ? 'Heute' : diffDays === 1 ? 'Morgen' : diffDays <= 7 ? `In ${diffDays} Tagen` : '';
      html += `<div class="appt-card">
        <div class="appt-date-box">
          <div class="appt-day">${d.getDate()}</div>
          <div class="appt-month">${months[d.getMonth()]}</div>
        </div>
        <div class="appt-info">
          <div class="appt-title">${appt.title}</div>
          <div class="appt-sub">${appt.sub || ''} ${appt.time ? '· ' + appt.time + ' Uhr' : ''}</div>
        </div>
        ${soonLabel ? `<div class="appt-soon">${soonLabel}</div>` : ''}
        ${showDelete ? `<button class="delete-btn" onclick="deleteAppt(${appt.id})">✕</button>` : ''}
      </div>`;
    });
    el.innerHTML = html;
  }

  // ---- RENDER ----
  function render() {
    renderEnergy();
    renderMeds('todayMeds', false);
    renderMeds('medsList', true);
    renderAppts('upcomingAppts', 3, false);
    renderAppts('apptsList', 999, true);
  }

  render();
</script>
